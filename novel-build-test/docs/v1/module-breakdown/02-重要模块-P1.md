### 2.3 数据模型

```sql
-- Agent提示词
CREATE TABLE agent_prompts (
    id              INTEGER PRIMARY KEY AUTOINCREMENT,
    agent_key       TEXT NOT NULL UNIQUE,
    content         TEXT NOT NULL,
    version         INTEGER DEFAULT 1,
    created_by      INTEGER REFERENCES users(id),
    created_at      DATETIME DEFAULT CURRENT_TIMESTAMP,
    updated_at      DATETIME DEFAULT CURRENT_TIMESTAMP
);

-- 提示词版本历史
CREATE TABLE prompt_versions (
    id              INTEGER PRIMARY KEY AUTOINCREMENT,
    agent_key       TEXT NOT NULL,
    version         INTEGER NOT NULL,
    content         TEXT NOT NULL,
    created_by      INTEGER REFERENCES users(id),
    created_at      DATETIME DEFAULT CURRENT_TIMESTAMP
);

-- 提示词模板
CREATE TABLE prompt_templates (
    id              INTEGER PRIMARY KEY AUTOINCREMENT,
    name            TEXT NOT NULL,
    category        TEXT,           -- 分类
    content         TEXT NOT NULL,
    is_preset       BOOLEAN DEFAULT FALSE,
    created_by      INTEGER REFERENCES users(id),
    created_at      DATETIME DEFAULT CURRENT_TIMESTAMP
);

-- Few-shot示例
CREATE TABLE fewshot_examples (
    id              INTEGER PRIMARY KEY AUTOINCREMENT,
    agent_key       TEXT NOT NULL,
    input           TEXT NOT NULL,
    output          TEXT NOT NULL,
    description     TEXT,
    sort_order      INTEGER DEFAULT 0,
    created_at      DATETIME DEFAULT CURRENT_TIMESTAMP
);
```

### 3.3 数据模型

```sql
-- LLM提供商
CREATE TABLE model_providers (
    id              INTEGER PRIMARY KEY AUTOINCREMENT,
    name            TEXT NOT NULL,
    type            TEXT NOT NULL,  -- openai/anthropic/deepseek/qwen/custom
    api_key         TEXT,           -- 加密存储
    base_url        TEXT,
    config          JSON,           -- 额外配置
    status          TEXT DEFAULT 'active',
    created_at      DATETIME DEFAULT CURRENT_TIMESTAMP,
    updated_at      DATETIME DEFAULT CURRENT_TIMESTAMP
);

-- 模型
CREATE TABLE models (
    id              INTEGER PRIMARY KEY AUTOINCREMENT,
    provider_id     INTEGER NOT NULL REFERENCES model_providers(id),
    name            TEXT NOT NULL,
    display_name    TEXT,
    max_tokens      INTEGER,
    input_price     REAL,           -- 输入价格/1K tokens
    output_price    REAL,           -- 输出价格/1K tokens
    status          TEXT DEFAULT 'active',
    created_at      DATETIME DEFAULT CURRENT_TIMESTAMP
);

-- 分级策略
CREATE TABLE model_tiers (
    id              INTEGER PRIMARY KEY AUTOINCREMENT,
    name            TEXT NOT NULL,  -- L1/L2/L3/L4/L5
    description     TEXT,
    rules           JSON NOT NULL,  -- 路由规则
    created_at      DATETIME DEFAULT CURRENT_TIMESTAMP,
    updated_at      DATETIME DEFAULT CURRENT_TIMESTAMP
);

-- 成本记录
CREATE TABLE cost_records (
    id              INTEGER PRIMARY KEY AUTOINCREMENT,
    execution_id    INTEGER REFERENCES executions(id),
    model_id        INTEGER NOT NULL REFERENCES models(id),
    input_tokens    INTEGER DEFAULT 0,
    output_tokens   INTEGER DEFAULT 0,
    cost            REAL DEFAULT 0,
    created_at      DATETIME DEFAULT CURRENT_TIMESTAMP
);
```

```sql
-- 去AI味配置
CREATE TABLE deai_config (
    id              INTEGER PRIMARY KEY AUTOINCREMENT,
    style           TEXT DEFAULT 'neutral',  -- formal/casual/neutral
    variety         JSON,                    -- 句式多样性配置
    humanize        JSON,                    -- 人性化配置
    intensity       TEXT DEFAULT 'medium',   -- off/light/medium/heavy
    updated_at      DATETIME DEFAULT CURRENT_TIMESTAMP
);

-- 领域术语
CREATE TABLE deai_terms (
    id              INTEGER PRIMARY KEY AUTOINCREMENT,
    term            TEXT NOT NULL,
    replacement     TEXT,           -- 替换词
    category        TEXT,           -- 分类
    created_at      DATETIME DEFAULT CURRENT_TIMESTAMP
);

-- 输出模板
CREATE TABLE deai_templates (
    id              INTEGER PRIMARY KEY AUTOINCREMENT,
    name            TEXT NOT NULL,
    pattern         TEXT NOT NULL,  -- 模板模式
    description     TEXT,
    created_at      DATETIME DEFAULT CURRENT_TIMESTAMP
);

-- 词频黑名单
CREATE TABLE deai_blacklist (
    id              INTEGER PRIMARY KEY AUTOINCREMENT,
    word            TEXT NOT NULL UNIQUE,
    replacement     TEXT,           -- 替换词
    created_at      DATETIME DEFAULT CURRENT_TIMESTAMP
);
```
